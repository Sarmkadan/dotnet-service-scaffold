# Caddy Configuration Example for dotnet-service-scaffold
#
# This file shows how to configure Caddy as a reverse proxy for
# dotnet-service-scaffold with HTTPS, compression, and security headers.
#
# Usage:
#   1. Copy this to /etc/caddy/Caddyfile
#   2. Run: sudo systemctl reload caddy
#   3. Access via: https://scaffold.example.com
#
# Reference: https://caddyserver.com/docs
#

# Basic configuration with automatic HTTPS
scaffold.example.com {
    reverse_proxy localhost:5000 {
        # Header configuration for proxied requests
        header_up X-Forwarded-For {http.request.remote.host}
        header_up X-Forwarded-Proto {http.request.proto}
        header_up X-Forwarded-Host {http.request.host}
        header_up X-Real-IP {http.request.remote.host}

        # Health check configuration
        health_uri /health
        health_interval 10s
        health_timeout 5s
        health_status 200

        # Connection settings
        timeout 30s
        keepalive 60s
    }

    # Enable gzip compression
    encode gzip

    # Logging
    log {
        output file /var/log/caddy/scaffold.log {
            roll_size 100mb
            roll_keep 10
            roll_keep_for 720h
        }
        level info
        format json
    }

    # Security headers
    header {
        # Enforce HTTPS
        Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"

        # Prevent MIME type sniffing
        X-Content-Type-Options "nosniff"

        # Prevent clickjacking
        X-Frame-Options "DENY"

        # XSS protection
        X-XSS-Protection "1; mode=block"

        # Referrer policy
        Referrer-Policy "strict-origin-when-cross-origin"

        # Permissions policy (formerly Feature-Policy)
        Permissions-Policy "geolocation=(), microphone=(), camera=()"
    }
}

# ============================================================================
# DEPLOYMENT INSTRUCTIONS
# ============================================================================
#
# 1. Install Caddy on Ubuntu/Debian:
#    sudo apt-get update
#    sudo apt-get install -y caddy
#
# 2. Create backup of existing config:
#    sudo cp /etc/caddy/Caddyfile /etc/caddy/Caddyfile.bak
#
# 3. Copy this configuration:
#    sudo cp caddy-example.txt /etc/caddy/Caddyfile
#
# 4. Edit the configuration:
#    sudo nano /etc/caddy/Caddyfile
#    - Change "scaffold.example.com" to your domain
#    - Update email address
#
# 5. Test configuration:
#    sudo caddy validate --config /etc/caddy/Caddyfile
#
# 6. Reload Caddy:
#    sudo systemctl reload caddy
#
# 7. Check status:
#    sudo systemctl status caddy
#
# 8. View logs:
#    sudo journalctl -u caddy -f
#
# ============================================================================
# MONITORING & TROUBLESHOOTING
# ============================================================================
#
# Check if service is running:
#   curl -v https://scaffold.example.com/health
#
# View Caddy logs:
#   sudo journalctl -u caddy -n 50 -f
#
# Reload configuration (graceful restart):
#   sudo systemctl reload caddy
#
# Completely restart:
#   sudo systemctl restart caddy
#
# Check certificate expiration:
#   openssl s_client -connect scaffold.example.com:443 -servername scaffold.example.com | grep -i "notAfter"
#
