# Caddy Configuration Example for dotnet-service-scaffold
#
# Caddy obtains and renews TLS certificates from Let's Encrypt automatically
# whenever you use a real domain name as the site address. No extra plugins or
# manual certificate management are required.
#
# QUICK START FOR AUTOMATIC HTTPS
# --------------------------------
# 1. Point your domain's A record to this server's IP address.
# 2. Set the ACME_EMAIL environment variable (used for expiry notifications):
#      export ACME_EMAIL="you@example.com"
#    Or hard-code it in the global block below.
# 3. Copy this file to /etc/caddy/Caddyfile, replace "scaffold.example.com"
#    with your real domain, then run: sudo systemctl reload caddy
#
# Caddy will automatically obtain a Let's Encrypt certificate on first request.
#
# Reference: https://caddyserver.com/docs/automatic-https
#

# ============================================================================
# GLOBAL OPTIONS — configure ACME email here or via ACME_EMAIL env var
# ============================================================================
{
    # Email address sent to Let's Encrypt for certificate expiry notifications.
    # Can be set via environment variable: export ACME_EMAIL="you@example.com"
    email {$ACME_EMAIL:admin@example.com}
}

# Basic configuration with automatic HTTPS
scaffold.example.com {
    reverse_proxy localhost:5000 {
        # Header configuration for proxied requests
        header_up X-Forwarded-For {http.request.remote.host}
        header_up X-Forwarded-Proto {http.request.proto}
        header_up X-Forwarded-Host {http.request.host}
        header_up X-Real-IP {http.request.remote.host}

        # Health check configuration
        health_uri /health
        health_interval 10s
        health_timeout 5s
        health_status 200

        # Connection settings
        timeout 30s
        keepalive 60s
    }

    # Enable gzip compression
    encode gzip

    # Logging
    log {
        output file /var/log/caddy/scaffold.log {
            roll_size 100mb
            roll_keep 10
            roll_keep_for 720h
        }
        level info
        format json
    }

    # Security headers
    header {
        # Enforce HTTPS
        Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"

        # Prevent MIME type sniffing
        X-Content-Type-Options "nosniff"

        # Prevent clickjacking
        X-Frame-Options "DENY"

        # XSS protection
        X-XSS-Protection "1; mode=block"

        # Referrer policy
        Referrer-Policy "strict-origin-when-cross-origin"

        # Permissions policy (formerly Feature-Policy)
        Permissions-Policy "geolocation=(), microphone=(), camera=()"
    }
}

# ============================================================================
# DEPLOYMENT INSTRUCTIONS
# ============================================================================
#
# 1. Install Caddy on Ubuntu/Debian:
#    sudo apt-get update
#    sudo apt-get install -y caddy
#
# 2. Set your ACME email (for Let's Encrypt expiry notifications):
#    sudo systemctl edit caddy
#    # Add under [Service]:
#    # Environment="ACME_EMAIL=you@example.com"
#
# 3. Create backup of existing config:
#    sudo cp /etc/caddy/Caddyfile /etc/caddy/Caddyfile.bak
#
# 4. Copy this configuration:
#    sudo cp caddy-example.txt /etc/caddy/Caddyfile
#
# 5. Edit the configuration:
#    sudo nano /etc/caddy/Caddyfile
#    - Change "scaffold.example.com" to your domain
#    - Update the ACME_EMAIL default value if not using the env var
#
# 6. Test configuration:
#    sudo caddy validate --config /etc/caddy/Caddyfile
#
# 7. Reload Caddy:
#    sudo systemctl reload caddy
#
# 8. Verify HTTPS is working (certificate may take a few seconds on first request):
#    curl -v https://scaffold.example.com/health
#
# 9. Check status:
#    sudo systemctl status caddy
#
# 10. View logs:
#    sudo journalctl -u caddy -f
#
# ============================================================================
# MONITORING & TROUBLESHOOTING
# ============================================================================
#
# Check if service is running:
#   curl -v https://scaffold.example.com/health
#
# View Caddy logs:
#   sudo journalctl -u caddy -n 50 -f
#
# Reload configuration (graceful restart):
#   sudo systemctl reload caddy
#
# Completely restart:
#   sudo systemctl restart caddy
#
# Check certificate expiration:
#   openssl s_client -connect scaffold.example.com:443 -servername scaffold.example.com | grep -i "notAfter"
#
